Get all user records sorted alphabetically

$customers = $db->all('select * from customers order by last_name, first_name;');


Get information on the last query

// display information about the last query, such as the SQL, errors, duration, etc $db->info(); // display the value of the variable for debugging // works especially well for displaying a 2-dimensional array with matching keys as a table $db->display($customers);


Get all user records sorted alphabetically

$customers = $db->all('select * from customers order by last_name, first_name;');


Get all info for the first user sorted alphabetically

$customers = $db->row('select * from customers order by last_name, first_name limit 1;');


Get the last name only for the first user sorted alphabetically

$lastName = $db->one('select last_name from customers order by last_name, first_name limit 1;');


Insert, Update, and Delete


Insert

$values = array('first_name'=>'Bob', 'last_name'=>'Jones'); $insertId = $db->insert('customers', $values);

Update first name of the newly added customer using the update function

$values = array('first_name' => 'Robert'); $isSuccess = $db->update('customers', $values, 'id=:id', array(':id' => $insertId));

Update last name of the newly added customer using the updateById function

$values = array('last_name' => 'Johnson'); $isSuccess = $db->updateById('customers', $values, $insertId);

Delete the newly added record

$isSuccess = $db->deleteById('customers', $insertId);
